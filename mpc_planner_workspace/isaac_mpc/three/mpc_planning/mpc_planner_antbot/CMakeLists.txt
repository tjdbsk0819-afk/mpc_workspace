cmake_minimum_required(VERSION 3.8)
project(mpc_planner_antbot)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
set(deps rclcpp ament_index_cpp nav_msgs std_msgs std_srvs antbot_interfaces mpc_planner ros_tools)
foreach(dep IN LISTS deps)
  find_package(${dep} REQUIRED)
endforeach()
add_executable(antbot_planner src/antbot_planner.cpp)
ament_target_dependencies(antbot_planner ${deps})
install(TARGETS antbot_planner DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
ament_package()
