cmake_minimum_required(VERSION 3.16)
if(POLICY CMP0100)
  cmake_policy(SET CMP0100 NEW)
endif()
project(antbot_gazebo_io)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(gz_sim_vendor REQUIRED)
find_package(gz_plugin_vendor REQUIRED)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tf2_msgs REQUIRED)
find_package(antbot_interfaces REQUIRED)
add_library(antbot_gazebo_io SHARED robot_io.cpp)
target_link_libraries(antbot_gazebo_io gz-sim8::gz-sim8 gz-plugin2::gz-plugin2)
ament_target_dependencies(antbot_gazebo_io rclcpp sensor_msgs nav_msgs tf2_msgs antbot_interfaces)

# Native Gazebo HUD: measured per-robot odometry, not command velocity.
set(CMAKE_AUTOMOC ON)
find_package(gz_gui_vendor REQUIRED)
find_package(gz-gui8 REQUIRED)
find_package(gz-transport13 REQUIRED)
find_package(gz-msgs10 REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Core Quick)
qt5_add_resources(SPEED_PANEL_RESOURCES speed_panel.qrc)
add_library(RobotSpeedPanel SHARED RobotSpeedPanel.hh RobotSpeedPanel.cc PathOverlay.cc ${SPEED_PANEL_RESOURCES})
target_link_libraries(RobotSpeedPanel gz-gui8::gz-gui8 Qt5::Core Qt5::Quick gz-plugin2::gz-plugin2 gz-transport13::gz-transport13 gz-msgs10::gz-msgs10)
ament_target_dependencies(RobotSpeedPanel rclcpp nav_msgs std_msgs)
